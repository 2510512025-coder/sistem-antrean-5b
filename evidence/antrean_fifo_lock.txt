## TC-02-P (positif): FIFO
$ cat data/antrean.txt
echo pekerjaan satu
echo pekerjaan dua
echo pekerjaan tiga
$ ./scripts/worker.sh
Memproses: echo pekerjaan satu
pekerjaan satu
Memproses: echo pekerjaan dua
pekerjaan dua
Memproses: echo pekerjaan tiga
pekerjaan tiga
Antrean kosong.
$ echo $?
0

## TC-02-N (negatif): worker kedua ditolak
$ ./scripts/worker.sh &
Memproses: echo pekerjaan satu
pekerjaan satu
Memproses: sleep 60
$ ./scripts/worker.sh
Worker lain sedang berjalan. Ditolak.
$ echo $?
1
Memproses: echo pekerjaan dua
pekerjaan dua
Antrean kosong.
